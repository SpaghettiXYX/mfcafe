    // AUDIO CHIME FOR RASPBERRY PI 5 SPEAKER

    function triggerChime() {

      try {

        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

        const osc = audioCtx.createOscillator();

        const gain = audioCtx.createGain();

        osc.type = "sine";

        osc.frequency.setValueAtTime(587.33, audioCtx.currentTime); // D5 note

        osc.frequency.exponentialRampToValueAtTime(880, audioCtx.currentTime + 0.3); // A5 note

        gain.gain.setValueAtTime(0.3, audioCtx.currentTime);

        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.8);

        osc.connect(gain);

        gain.connect(audioCtx.destination);

        osc.start();

        osc.stop(audioCtx.currentTime + 0.8);

      } catch (e) {

        console.log("Audio not supported or restricted by browser policy");

      }

    }

              <span class="text-xs font-bold px-2.5 py-1 rounded-full bg-emerald-100 text-emerald-800 elapsed-timer" data-time="${order.orderTime}">

                ${formatTimer(elapsedSec)}

              </span>

            </div>

            <div class="border-t border-b py-3 my-2">

              <p class="font-extrabold text-slate-800 text-lg">${order.productName}</p>

              <p class="text-xs font-bold ${order.temp === 'Hot' ? 'text-rose-600' : 'text-blue-600'}">${order.temp.toUpperCase()}</p>

            Call Drink Ready (#${order.orderNum})

          </button>

        `;

        container.appendChild(card);

      });

    }


    function markOrderReady(orderNum) {

      const order = state.orders.find(o => o.orderNum === orderNum);

      if (order) {

        order.status = "READY";

        order.pickupTime = new Date().toISOString();

        triggerChime(); // Sound notification

        renderAll();

      }

    }

    // 4. PORTABLE CLEARING FUNCTIONS

    function renderClearPage() {

      const container = document.getElementById('clearQuickButtons');

      container.innerHTML = "";

      const readyOrders = state.orders.filter(o => o.status === "READY");


      readyOrders.forEach(order => {

        const btn = document.createElement('button');

        btn.className = "bg-amber-100 hover:bg-amber-200 text-amber-900 font-extrabold text-lg px-4 py-2 rounded-xl border border-amber-300";

        btn.innerText = `#${order.orderNum}`;

        btn.onclick = () => clearOrder(order.orderNum);

        container.appendChild(btn);

      });

    }


    function clearOrder(orderNum) {

      const order = state.orders.find(o => o.orderNum === String(orderNum));

      if (order) {

        order.status = "CLEARED";

        renderAll();

      }

    }


    function submitManualClear() {

      const val = document.getElementById('clearInputNum').value;

      if (val) {

        clearOrder(val);

        document.getElementById('clearInputNum').value = "";

      }

    }

    function toggleMenuAvailability(id) {

      const item = state.menu.find(m => m.id === id);

      if (item) {

        item.available = !item.available;

        renderAll();

      }

    }


    // TIMER HELPERS

    function updateTimers() {

      document.querySelectorAll('.elapsed-timer').forEach(el => {

        const orderTime = new Date(el.getAttribute('data-time'));

        const elapsedSec = Math.floor((new Date() - orderTime) / 1000);

        el.innerText = formatTimer(elapsedSec);

      });

    }


    function formatTimer(sec) {

      const m = Math.floor(sec / 60);

      const s = sec % 60;

      return `${m}:${s < 10 ? '0' : ''}${s}`;

    }
